# 错误处理机制

## 错误处理概述

Go 语言采用显式的错误处理机制，与 Java 的异常处理和 JavaScript 的 try-catch 不同。

```
Go 错误处理
├── error 接口
│   └── 内置接口，只有一个 Error() 方法
├── errors 包
│   └── 提供创建和操作错误的函数
├── 自定义错误类型
│   └── 实现 error 接口的结构体
└── panic 和 recover
    └── 类似于异常，但使用较少
```

## 基本 error 处理

```go
package main

import (
	"errors"
	"fmt"
	"os"
)

func main() {
	// 1. 检查错误
	result, err := divide(10, 2)
	if err != nil {
		fmt.Println("错误:", err)
	} else {
		fmt.Println("结果:", result)
	}

	// 2. 处理错误
	result, err = divide(10, 0)
	if err != nil {
		fmt.Println("除法错误:", err)
	}

	// 3. 创建错误
	err = errors.New("自定义错误")
	fmt.Println("错误:", err)

	// 4. 格式化错误
	err = fmt.Errorf("错误码 %d: %s", 404, "未找到")
	fmt.Println("格式化错误:", err)
}

func divide(a, b float64) (float64, error) {
	if b == 0 {
		return 0, errors.New("除数不能为零")
	}
	return a / b, nil
}
```

## 自定义错误类型

```go
package main

import "fmt"

// 自定义错误类型
type ValidationError struct {
	Field   string
	Message string
}

func (e *ValidationError) Error() string {
	return fmt.Sprintf("字段 %s 验证失败: %s", e.Field, e.Message)
}

// 另一种自定义错误类型
type HTTPError struct {
	StatusCode int
	Message    string
}

func (e *HTTPError) Error() string {
	return fmt.Sprintf("HTTP %d: %s", e.StatusCode, e.Message)
}

func main() {
	err := validateAge(-5)
	if err != nil {
		fmt.Println("验证错误:", err)

		// 类型断言
		if validationErr, ok := err.(*ValidationError); ok {
			fmt.Printf("字段: %s\n", validationErr.Field)
			fmt.Printf("消息: %s\n", validationErr.Message)
		}
	}

	err = fetchResource("/not-found")
	if err != nil {
		fmt.Println("HTTP 错误:", err)

		if httpErr, ok := err.(*HTTPError); ok {
			fmt.Printf("状态码: %d\n", httpErr.StatusCode)
		}
	}
}

func validateAge(age int) error {
	if age < 0 {
		return &ValidationError{
			Field:   "age",
			Message: "年龄不能为负数",
		}
	}
	if age > 150 {
		return &ValidationError{
			Field:   "age",
			Message: "年龄不能超过150",
		}
	}
	return nil
}

func fetchResource(path string) error {
	if path == "/not-found" {
		return &HTTPError{
			StatusCode: 404,
			Message:    "资源未找到",
		}
	}
	return nil
}
```

## 错误包装

```go
package main

import (
	"errors"
	"fmt"
)

// 错误包装（Go 1.13+）
func main() {
	// 1. 基本包装
	err := readFile()
	if err != nil {
		fmt.Println("错误:", err)

		// 检查是否是特定错误
		if errors.Is(err, os.ErrNotExist) {
			fmt.Println("文件不存在")
		}
	}

	// 2. 解包错误
	err = process()
	if err != nil {
		fmt.Println("原始错误:", errors.Unwrap(err))
	}

	// 3. 多层包装
	err = multiLevel()
	if err != nil {
		fmt.Println("错误链:", err)

		// 检查错误链中是否包含特定错误
		if errors.Is(err, os.ErrNotExist) {
			fmt.Println("错误链包含文件不存在错误")
		}
	}
}

func readFile() error {
	_, err := os.ReadFile("nonexistent.txt")
	if err != nil {
		// 包装错误，添加上下文
		return fmt.Errorf("读取配置文件失败: %w", err)
	}
	return nil
}

func process() error {
	err := readFile()
	if err != nil {
		return fmt.Errorf("处理阶段失败: %w", err)
	}
	return nil
}

func multiLevel() error {
	err := process()
	if err != nil {
		return fmt.Errorf("最终处理失败: %w", err)
	}
	return nil
}
```

## 错误值比较

```go
package main

import (
	"errors"
	"fmt"
)

// 定义可比较的错误值
var (
	ErrNotFound   = errors.New("not found")
	ErrPermission = errors.New("permission denied")
)

func main() {
	// 1. 使用 errors.Is 比较错误
	err := lookupResource("item1")
	if errors.Is(err, ErrNotFound) {
		fmt.Println("资源未找到")
	}

	// 2. 自定义错误类型的比较
	err = getUser(0)
	if err != nil {
		var notFoundErr *NotFoundError
		if errors.As(err, &notFoundErr) {
			fmt.Printf("未找到用户 ID: %d\n", notFoundErr.ID)
		}
	}

	// 3. 直接比较错误变量
	err = doSomething()
	if err == ErrPermission {
		fmt.Println("权限不足")
	}
}

type NotFoundError struct {
	ID int
}

func (e *NotFoundError) Error() string {
	return fmt.Sprintf("ID %d not found", e.ID)
}

func lookupResource(id string) error {
	if id == "notexists" {
		return ErrNotFound
	}
	return nil
}

func getUser(id int) error {
	if id == 0 {
		return &NotFoundError{ID: id}
	}
	return nil
}

func doSomething() error {
	return ErrPermission
}
```

## Panic 和 Recover

```go
package main

import (
	"fmt"
)

// panic 和 recover 类似于异常，但应谨慎使用
func main() {
	// 1. recover 捕获 panic
	defer func() {
		if r := recover(); r != nil {
			fmt.Println("捕获 panic:", r)
		}
	}()

	// 2. 触发 panic
	// panic("发生严重错误！")

	// 3. 在函数中使用 recover
	safeFunction()

	// 4. defer 执行顺序
	deferOrder()
}

func safeFunction() {
	defer func() {
		if r := recover(); r != nil {
			fmt.Println("safeFunction 捕获:", r)
		}
	}()

	fmt.Println("safeFunction 开始")
	panic("safeFunction 中发生 panic")
}

func deferOrder() {
	defer fmt.Println("defer 1")
	defer fmt.Println("defer 2")
	defer fmt.Println("defer 3")

	fmt.Println("函数执行")
}
```

## 错误处理最佳实践

```go
package main

import (
	"errors"
	"fmt"
	"log"
	"os"
)

// 1. 尽早返回
func goodErrorHandling(data string) (result string, err error) {
	if data == "" {
		return "", errors.New("数据不能为空")
	}

	if len(data) > 100 {
		return "", errors.New("数据过长")
	}

	return "处理: " + data, nil
}

// 2. 错误上下文要清晰
func readConfig() error {
	config, err := os.ReadFile("config.yaml")
	if err != nil {
		return fmt.Errorf("读取配置文件失败: %w", err)
	}

	// 处理配置
	_ = config
	return nil
}

// 3. 使用 sentinel 错误
var (
	ErrInvalidInput = errors.New("invalid input")
	ErrNotFound     = errors.New("not found")
)

func validate(input string) error {
	if input == "" {
		return ErrInvalidInput
	}
	return nil
}

// 4. 错误日志记录
func processWithError() error {
	err := someOperation()
	if err != nil {
		log.Printf("操作失败: %v", err)
		return fmt.Errorf("处理失败: %w", err)
	}
	return nil
}

// 5. 使用结构体包含错误详情
type DetailedError struct {
	Op  string
	Err error
}

func (e *DetailedError) Error() string {
	return fmt.Sprintf("%s: %v", e.Op, e.Err)
}

func (e *DetailedError) Unwrap() error {
	return e.Err
}

func detailedOperation() error {
	err := someOperation()
	if err != nil {
		return &DetailedError{
			Op:  "detailedOperation",
			Err: err,
		}
	}
	return nil
}

func someOperation() error {
	return errors.New("模拟错误")
}

func main() {
	// 示例：尽早返回
	result, err := goodErrorHandling("test")
	if err != nil {
		fmt.Println("错误:", err)
	} else {
		fmt.Println(result)
	}

	// 示例：sentinel 错误
	err = validate("")
	if errors.Is(err, ErrInvalidInput) {
		fmt.Println("输入无效")
	}

	// 示例：详细错误
	err = detailedOperation()
	if err != nil {
		fmt.Println("详细错误:", err)
		if opErr := (*DetailedError)(nil); errors.As(err, &opErr) {
			fmt.Printf("操作: %s\n", opErr.Op)
		}
	}
}
```

## 与 Java/JavaScript 对比

| Go | Java | JavaScript | 说明 |
|----|------|------------|------|
| `if err != nil {}` | `try { } catch (Exception e) {}` | `try { } catch (err) {}` | Go 显式检查 |
| `errors.New()` | `new Exception()` | `new Error()` | 创建错误 |
| `fmt.Errorf("msg: %w", err)` | `new RuntimeException("msg", e)` | `new Error("msg")` | 包装错误 |
| `errors.Is(err, target)` | `e instanceof Type` | `e instanceof Error` | 错误比较 |
| `panic()` | `throw e` | `throw e` | 抛出异常 |
| `recover()` | `catch` 块 | `catch` 块 | 捕获异常 |
| 返回 error | 抛出异常 | 抛出异常 | 处理方式 |
