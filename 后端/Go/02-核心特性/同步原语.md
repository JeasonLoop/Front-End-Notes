# 同步原语

## 同步原语概述

Go 提供了多种同步原语用于协调 goroutine 之间的访问和操作。

```
Go 同步原语
├── sync.Mutex
│   └── 互斥锁，保护共享资源
├── sync.RWMutex
│   └── 读写锁，允许多读或单写
├── sync.WaitGroup
│   └── 等待一组 goroutine 完成
├── sync.Once
│   └── 确保函数只执行一次
├── sync.Cond
│   └── 条件变量，等待/通知机制
├── sync.Map
│   └── 并发安全的 Map
└── atomic
    └── 原子操作
```

## Mutex（互斥锁）

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

type Counter struct {
	mu    sync.Mutex
	value int
}

func (c *Counter) Increment() {
	c.mu.Lock()         // 获取锁
	defer c.mu.Unlock() // 确保释放锁

	c.value++
}

func (c *Counter) Get() int {
	c.mu.Lock()
	defer c.mu.Unlock()

	return c.value
}

func main() {
	counter := &Counter{}
	var wg sync.WaitGroup

	// 启动多个 goroutine 并发增加计数器
	for i := 0; i < 100; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			counter.Increment()
		}()
	}

	wg.Wait()
	fmt.Println("最终计数:", counter.Get())
}

// 避免死锁示例
func avoidDeadlock() {
	var mu sync.Mutex

	// 错误：重复加锁会导致死锁
	// mu.Lock()
	// mu.Lock() // 死锁！

	// 正确：确保成对使用 Lock/Unlock
	mu.Lock()
	defer mu.Unlock()

	// 临界区代码
	time.Sleep(100 * time.Millisecond)
}
```

## RWMutex（读写锁）

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

type SafeMap struct {
	mu   sync.RWMutex
	data map[string]int
}

func NewSafeMap() *SafeMap {
	return &SafeMap{
		data: make(map[string]int),
	}
}

func (m *SafeMap) Read(key string) (int, bool) {
	m.mu.RLock()         // 读锁
	defer m.mu.RUnlock()

	val, ok := m.data[key]
	return val, ok
}

func (m *SafeMap) Write(key string, value int) {
	m.mu.Lock()          // 写锁
	defer m.mu.Unlock()

	m.data[key] = value
}

func main() {
	sm := NewSafeMap()
	var wg sync.WaitGroup

	// 初始化数据
	sm.Write("a", 1)
	sm.Write("b", 2)
	sm.Write("c", 3)

	// 启动多个读操作（可以并发）
	for i := 0; i < 5; i++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			if val, ok := sm.Read("a"); ok {
				fmt.Printf("读操作%d: a = %d\n", id, val)
			}
			time.Sleep(100 * time.Millisecond)
		}(i)
	}

	// 启动写操作（互斥）
	for i := 0; i < 3; i++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			key := fmt.Sprintf("key%d", id)
			sm.Write(key, id*10)
			fmt.Printf("写操作%d: 写入 %s = %d\n", id, key, id*10)
		}(i)
	}

	wg.Wait()
	fmt.Println("所有操作完成")
}
```

## WaitGroup

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

func main() {
	var wg sync.WaitGroup

	// 添加等待的 goroutine 数量
	for i := 1; i <= 5; i++ {
		wg.Add(1)
		go worker(i, &wg)
	}

	// 等待所有 goroutine 完成
	wg.Wait()
	fmt.Println("所有 worker 完成")
}

func worker(id int, wg *sync.WaitGroup) {
	defer wg.Done() // 完成时调用

	fmt.Printf("Worker %d 开始工作\n", id)
	time.Sleep(time.Duration(id) * 100 * time.Millisecond)
	fmt.Printf("Worker %d 完成工作\n", id)
}

// WaitGroup 嵌套示例
func nestedWaitGroup() {
	var outerWg sync.WaitGroup

	for i := 1; i <= 3; i++ {
		outerWg.Add(1)
		go func(taskID int) {
			defer outerWg.Done()

			var innerWg sync.WaitGroup
			for j := 1; j <= 2; j++ {
				innerWg.Add(1)
				go func(subID int) {
					defer innerWg.Done()
					time.Sleep(100 * time.Millisecond)
					fmt.Printf("任务 %d-%d 完成\n", taskID, subID)
				}(j)
			}
			innerWg.Wait()
		}(i)
	}

	outerWg.Wait()
	fmt.Println("所有任务完成")
}
```

## Once

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

type Database struct {
	connected bool
}

func (db *Database) Connect() {
	if db.connected {
		fmt.Println("数据库已连接，跳过连接")
		return
	}

	fmt.Println("正在连接数据库...")
	time.Sleep(1 * time.Second)
	db.connected = true
	fmt.Println("数据库连接成功")
}

type SafeDatabase struct {
	mu      sync.Mutex
	once    sync.Once
	db      *Database
}

func (sdb *SafeDatabase) GetConnection() *Database {
	sdb.once.Do(func() {
		fmt.Println("初始化数据库连接...")
		sdb.db = &Database{}
		sdb.db.Connect()
	})
	return sdb.db
}

func main() {
	var wg sync.WaitGroup
	sdb := &SafeDatabase{}

	// 多个 goroutine 同时尝试获取连接
	for i := 0; i < 5; i++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			conn := sdb.GetConnection()
			fmt.Printf("Goroutine %d 获取到连接\n", id)
		}(i)
	}

	wg.Wait()
}

// 单例模式示例
type Singleton struct {
	data string
}

var (
	instance *Singleton
	once     sync.Once
)

func GetSingleton() *Singleton {
	once.Do(func() {
		instance = &Singleton{data: "Singleton Instance"}
		fmt.Println("创建单例")
	})
	return instance
}
```

## Cond（条件变量）

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

func main() {
	// 生产者-消费者模式使用条件变量
	var mu sync.Mutex
	cond = sync.NewCond(&mu)
	buffer = make([]int, 0, 10)

	// 启动生产者
	for i := 1; i <= 3; i++ {
		go producer(i)
	}

	// 启动消费者
	for i := 1; i <= 2; i++ {
		go consumer(i)
	}

	// 运行一段时间后停止
	time.Sleep(5 * time.Second)
	fmt.Println("主程序结束")
}

var (
	cond   *sync.Cond
	buffer []int
)

func producer(id int) {
	for {
		cond.L.Lock()

		// 缓冲区满时等待
		for len(buffer) >= 10 {
			fmt.Printf("生产者%d: 缓冲区满，等待...\n", id)
			cond.Wait()
		}

		// 生产数据
		item := id*10 + len(buffer)
		buffer = append(buffer, item)
		fmt.Printf("生产者%d: 生产 %d，缓冲区大小 %d\n", id, item, len(buffer))

		// 通知消费者
		cond.Signal()
		cond.L.Unlock()

		time.Sleep(200 * time.Millisecond)
	}
}

func consumer(id int) {
	for {
		cond.L.Lock()

		// 缓冲区空时等待
		for len(buffer) == 0 {
			fmt.Printf("消费者%d: 缓冲区空，等待...\n", id)
			cond.Wait()
		}

		// 消费数据
		item := buffer[0]
		buffer = buffer[1:]
		fmt.Printf("消费者%d: 消费 %d，缓冲区大小 %d\n", id, item, len(buffer))

		// 通知生产者
		cond.Signal()
		cond.L.Unlock()

		time.Sleep(300 * time.Millisecond)
	}
}
```

## Sync.Map

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

func main() {
	var m sync.Map
	var wg sync.WaitGroup

	// 并发写入
	for i := 0; i < 100; i++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			key := fmt.Sprintf("key%d", id%10)
			m.Store(key, id)
		}(i)
	}

	// 并发读取
	for i := 0; i < 50; i++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			key := fmt.Sprintf("key%d", id%10)
			if value, ok := m.Load(key); ok {
				fmt.Printf("读取 %s: %v\n", key, value)
			}
		}(i)
	}

	wg.Wait()

	// 遍历 Map
	fmt.Println("\nMap 内容:")
	m.Range(func(key, value any) bool {
		fmt.Printf("  %s: %v\n", key, value)
		return true
	})

	// 删除键
	m.Delete("key0")

	// 加载或存储
	actual, loaded := m.LoadOrStore("newKey", 999)
	fmt.Printf("\n加载或存储: newKey = %v, 已存在: %v\n", actual, loaded)
}
```

## Atomic（原子操作）

```go
package main

import (
	"fmt"
	"sync"
	"sync/atomic"
)

func main() {
	// 原子计数器
	var counter int64
	var wg sync.WaitGroup

	for i := 0; i < 1000; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			atomic.AddInt64(&counter, 1)
		}()
	}

	wg.Wait()
	fmt.Println("原子计数:", atomic.LoadInt64(&counter))

	// 原子读取和写入
	var value int64
	atomic.StoreInt64(&value, 42)
	fmt.Println("存储后读取:", atomic.LoadInt64(&value))

	// 原子比较交换
	var x int64 = 10
	swapped := atomic.CompareAndSwapInt64(&x, 10, 20)
	fmt.Printf("CAS: x=%d, swapped=%v\n", x, swapped)

	// 原子交换
	var y int64 = 100
	old := atomic.SwapInt64(&y, 200)
	fmt.Printf("交换: old=%d, new=%d\n", old, y)

	// 原子指针操作
	type Config struct {
		Timeout int
	}
	var config atomic.Value
	config.Store(&Config{Timeout: 30})

	loaded := config.Load().(*Config)
	fmt.Printf("配置超时: %d\n", loaded.Timeout)

	// 无锁栈示例
	stack := NewLockFreeStack()
	for i := 0; i < 10; i++ {
		stack.Push(i)
	}

	for {
		val := stack.Pop()
		if val == nil {
			break
		}
		fmt.Println("弹出:", val)
	}
}

type Node struct {
	value int
	next  *atomic.Pointer[Node]
}

type LockFreeStack struct {
	head atomic.Pointer[Node]
}

func (s *LockFreeStack) Push(value int) {
	newNode := &Node{value: value}
	newNode.next.Store(s.head.Load())

	for {
		oldHead := s.head.Load()
		newNode.next.Store(oldHead)
		if s.head.CompareAndSwap(oldHead, newNode) {
			break
		}
	}
}

func (s *LockFreeStack) Pop() interface{} {
	for {
		oldHead := s.head.Load()
		if oldHead == nil {
			return nil
		}

		next := oldHead.next.Load()
		if s.head.CompareAndSwap(oldHead, next) {
			return oldHead.value
		}
	}
}
```

## 同步最佳实践

```go
package main

import (
	"sync"
)

// 1. 使用 defer 确保锁释放
func goodLockUsage(mu *sync.Mutex) {
	mu.Lock()
	defer mu.Unlock()
	// 临界区代码
}

// 2. 避免锁的嵌套
// bad: 可能导致死锁
func badNestedLock(mu1, mu2 *sync.Mutex) {
	mu1.Lock()
	mu2.Lock()
	// ...
	mu2.Unlock()
	mu1.Unlock()
}

// good: 固定加锁顺序
func goodNestedLock(mu1, mu2 *sync.Mutex) {
	// 始终按相同顺序获取锁
	mu1.Lock()
	defer mu1.Unlock()

	mu2.Lock()
	defer mu2.Unlock()
	// ...
}

// 3. 读写锁按需使用
type Counter struct {
	mu sync.RWMutex
	v  int
}

func (c *Counter) Get() int {
	c.mu.RLock()
	defer c.mu.RUnlock()
	return c.v
}

func (c *Counter) Set(v int) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.v = v
}

// 4. 使用 channel 优先于 Mutex
// channel 更符合 Go 的通信理念
func channelStyle() {
	ch := make(chan int, 1)
	go func() {
		ch <- 42
	}()
	val := <-ch
	_ = val
}

func main() {
	var mu sync.Mutex
	goodLockUsage(&mu)
}
```

## 与 Java 对比

| Go | Java | 说明 |
|----|------|------|
| `sync.Mutex` | `ReentrantLock` | 功能相似 |
| `sync.RWMutex` | `ReadWriteLock` | 读写锁 |
| `sync.WaitGroup` | `CountDownLatch` | 等待机制 |
| `sync.Once` | 单例模式 | Go 原生支持 |
| `sync.Cond` | `Condition` | 条件变量 |
| `atomic` | `Atomic*` 类 | 原子操作 |
| `sync.Map` | `ConcurrentHashMap` | 并发 Map |
